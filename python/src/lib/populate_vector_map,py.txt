from lib.eebo_logging import logger
from lib.eebo_db import get_connection
from canon_stream import canonical_token_stream


def populate_vector_map(conn):
    inserted = 0

    with conn.cursor() as cur:

        for doc_id, token_idx, _ in canonical_token_stream(conn):

            cur.execute("""
                INSERT INTO vector_map (doc_id, token_idx)
                VALUES (%s, %s)
                ON CONFLICT (doc_id, token_idx) DO NOTHING
            """, (doc_id, token_idx))

            inserted += 1

            if inserted % 100_000 == 0:
                conn.commit()
                logger.info(f"[vector_map] inserted={inserted}")

        conn.commit()

    logger.info(f"[vector_map COMPLETE] {inserted}")
